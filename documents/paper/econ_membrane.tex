\ExplSyntaxOn
\tl_new:N \l_ftoo_tagdep_tl
\cs_new_protected:Npn \ftoo_econ_process:
  {
    \tl_clear:N \l_ftoo_tagdep_tl
    \clist_set:NV \l_tmpa_clist \ftooEconDeparts
    \clist_if_empty:NT \l_tmpa_clist
      { \msg_error:nn { ftoo } { econ-no-departure } }
    \int_compare:nNnF { \clist_count:N \l_tmpa_clist } = { 1 }
      { \msg_error:nnx { ftoo } { econ-one-departure-only }
          { \clist_use:Nn \l_tmpa_clist { ,~ } } }
    \clist_map_inline:Nn \l_tmpa_clist
      {
        \ftoo_roster_admits:NnF \g_ftoo_departures_clist {##1}
          { \msg_error:nnn { ftoo } { bad-departure } {##1} }
        \tl_if_empty:NF \l_ftoo_tagdep_tl { \tl_put_right:Nn \l_ftoo_tagdep_tl { ,~ } }
        \tl_put_right:Nn \l_ftoo_tagdep_tl { \ftoo_tag_bare:n {d:##1} }
        \ftoo_write_row:nVVVV {##1} \ftooEconLabel \ftooEconRef \ftooEconCites \ftooEconStopsat
      }
  }

  { An~[ECON]~box~declares~EXACTLY~ONE~departure;~got~'#1'.~A~box~that~departs~
    two~ways~has~not~been~adjudicated~---~pick~the~principal~move. }

\NewDocumentCommand{\ftooBoxTitle}{m m m m m}
  {
    #1 \nobreakspace #2 .
    \tl_if_blank:eF {#3} { \nobreakspace #3 }
    \tl_if_blank:eF {#5}
      {
        \nobreakspace
        { \normalfont \mdseries \footnotesize ( #4 \nobreakspace \ftoo_tag_located:n {#5} ) }
      }
  }

  { \tl_if_blank:eF {#1} { \par \medskip \hfill { \footnotesize \textsc { #1 } } } }
