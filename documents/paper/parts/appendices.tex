\section{Recovery}
\label{app:recovery}
\auditnote{The recovery appendix quantifies the null-verdict wall. Its estimates are about a
single $\anchormetric$-scalar externality and do not create a portable certification.}

The estimand is a scalar externality recovered on one anchor:
\begin{align}
  &\recoveryfact &=& \ \normG{\outcomechange{k}}
    \quad \dimwelfare
    &:& \eqtag{recov:estimand}{estimand}
\end{align}

\begin{remark}[\hdr{recov:factor}]
  Reports satisfy
  \begin{equation}
    \reports=\incidence\recoveryfact+\frameloading\frameoffsets+\residuals.
    \label{eq:recov:reports}
  \end{equation}
  For a single-subject certificate, $\incidence=\mathbf 1$. The residual covariance is
  \begin{equation}
    \reportcov=\idcov+\frameloading\anchorprior\frameloading\transpose,
    \label{eq:recov:covsplit}
  \end{equation}
  with $\idcov,\anchorprior,\reportcov:\dimpublicwelfaresq$. The GLS precision is
  \begin{equation}
    \precisionof{\recoveryfact}=\mathbf 1\transpose\reportcov\sp{-1}\mathbf 1
    \qquad \dimpublicwelfareinvsq.
    \label{eq:recov:gls}
  \end{equation}
  The split $\reportcov=\idcov+\frameloading\anchorprior\frameloading\transpose$ is
  anchor-relative: the data alone do not decide which correlation is idiosyncratic and which
  is common-mode.
\end{remark}

\begin{proposition}[\hdr{recov:sat}]\grade{feas}
  Set
  \begin{align}
    \woodburyJ &= \mathbf 1\transpose\idcov\sp{-1}\mathbf 1
      \label{eq:recov:woodJ}\\
    \woodburyd &= \frameloading\transpose\idcov\sp{-1}\mathbf 1
      \label{eq:recov:woodd}\\
    \framegram &= \frameloading\transpose\idcov\sp{-1}\frameloading.
      \label{eq:recov:woodG}
  \end{align}
  Then Woodbury gives
  \begin{equation}
    \boxed{
      \precisionof{\recoveryfact}
      =\woodburyJ-\woodburyd\transpose
      (\anchorprior\sp{-1}+\framegram)\sp{-1}\woodburyd.
    }
    \label{eq:recov:sat}
  \end{equation}
  For one routed block,
  \begin{equation}
    \precision\sb{\ell}
    =\frac{\mathbf 1\transpose\idcov\sb{\ell}\sp{-1}\mathbf 1}
      {1+\anchorvariance\,\mathbf 1\transpose\idcov\sb{\ell}\sp{-1}\mathbf 1}.
    \label{eq:recov:block}
  \end{equation}
  Define
  \begin{align}
    \cleanreads[\ell]
      &:=\idvariance\,\mathbf 1\transpose\idcov\sb{\ell}\sp{-1}\mathbf 1
      \qquad \dimcount
      \label{eq:recov:cleanreads}\\
    \infopool[\ell]
      &:=\frac{\idvariance}{\anchorvariance}
      \qquad \dimcount.
      \label{eq:recov:infopool}
  \end{align}
  Then effective reads saturate at the anchor information pool:
  \begin{align}
    &\effectivereads[\ell] &=& \ \frac{\cleanreads[\ell]}{1+\cleanreads[\ell]/\infopool[\ell]}
      &:& \eqtag{recov:effreads}{saturation}\\
    &\precision\sb{\ell} &=& \ \frac{\effectivereads[\ell]}{\idvariance}
      \label{eq:recov:effprecision}
  \end{align}
  Readers buy down idiosyncrasy; anchors set the cap.
\end{proposition}

The covering condition carries its own handle, \handle{recov:cover}: the frames span the
constant vector, $\mathbf 1\in\operatorname{col}(\frameloading)$.

\begin{proposition}[\hdr{recov:blocks}]\grade{feas}
  For orthogonal routed blocks,
  \begin{equation}
    \precisionof{\recoveryfact}
    =\sum_\ell\frac{\effectivereads[\ell]}{\idvariance}
    \to
    \sum_\ell\frac{1}{\anchorvariance}.
    \label{eq:recov:blocksum}
  \end{equation}
  For overlapping frames, under the covering condition
  (\ftag*{recov:cover}) $\mathbf 1\in\operatorname{col}(\frameloading)$,
  \begin{equation}
    \boxed{
      \precision\sb{\max}
      =\min_{\coeffvec:\frameloading\coeffvec=\mathbf 1}
        \coeffvec\transpose\anchorprior\sp{-1}\coeffvec
      =\mathbf 1\transpose
        \pinv{\frameloading\anchorprior\frameloading\transpose}\mathbf 1.
    }
    \label{eq:recov:ceiling}
  \end{equation}
  If $\mathbf 1\notin\operatorname{col}(\frameloading)$, some component is not
  anchor-confounded and precision can grow without the finite ceiling.
\end{proposition}

\begin{remark}[\hdr{recov:span}]
  In the one-colour linear corner
  \begin{equation}
    \witnessreport{w}
      =
      \recoveryfact+\colourgain{w}\colouramplitude+\spannoise{w},
    \label{eq:recov:spanmodel}
  \end{equation}
  recovery of $\recoveryfact$ is governed by $\colourspan$. The covering condition
  $\onesvector\in\operatorname{col}(\frameloadingint)$ is the one-colour confound in this
  corner: a nonzero constant $\colourgain{w}$ makes the intercept and colour-amplitude
  indistinguishable. Opposed colours, or the zero-colour corner, remove that particular
  confound. Heterogeneous stakes replace $\colourgain{w}$ by the GLS loading proportional to
  the witness's colour-loading and recover the saturation law \ftag{recov:sat}. Thus
  $\colourspan$ is the quantitative face of the block-orthogonality named by \ftag{recov:blocks},
  not a replacement for the full covariance model.
\end{remark}

\begin{insight}[\hdr{recov:ext}]\grade{feas}
  Independent anchors are extensive for truth; readers are intensive once the cap binds.
\end{insight}

\begin{proposition}[\hdr{recov:route}]\grade{feas}
  Effective routed clean reads are
  \begin{equation}
    \cleanreads[\ell]
    =\idvariance\,\mathbf 1\transpose\idcov\sb{\ell}\sp{-1}\mathbf 1.
    \label{eq:recov:route}
  \end{equation}
  In a concrete corner,
  \begin{equation}
    \cleanreads[\ell]=\sum_w\eta_w q_{\ell w},
    \qquad
    \sum_\ell q_{\ell w}\le1.
    \label{eq:recov:routecorner}
  \end{equation}
\end{proposition}

\begin{insight}[\hdr{recov:split}]\grade{feas}
  Credibility routes. Unbias and independence weigh:
  \begin{align}
    \text{credibility} &\Rightarrow \text{routing}
      \label{eq:recov:split-cred}\\
    \text{unbias + independence} &\Rightarrow \idcov\sb{\ell}.
      \label{eq:recov:split-unbias}
  \end{align}
\end{insight}

\begin{remark}[\hdr{recov:Lhat}]
  \begin{align}
    \audiencediversity
      &=\frac{\left(\sum_i s_i(Q)\right)^2}{\sum_i s_i(Q)^2}
      \label{eq:recov:lhat}\\
    Q_{rw} &=\positivepart{\lean{r}{w}}.
      \label{eq:recov:Q}
  \end{align}
  But $\audiencediversity\neq\text{true anchor count}$.
\end{remark}

\begin{theorem}[\hdr{recov:null}]\grade{nec}
  Let $\frameloadingint$ be internally pinned frames. If
  \begin{equation}
    \mathbf 1\in\operatorname{col}(\frameloadingint),
    \label{eq:recov:covering}
  \end{equation}
  then there is $\coeffvec$ with $\frameloadingint\coeffvec=\mathbf 1$. The transformation
  \begin{equation}
    (\recoveryfact,\frameoffsets)
    \mapsto
    (\recoveryfact+t,\frameoffsets-t\coeffvec)
    \label{eq:recov:reparam}
  \end{equation}
  leaves
  \begin{equation}
    \reports=\mathbf 1\recoveryfact+\frameloadingint\frameoffsets+\residuals
    \label{eq:recov:invariant}
  \end{equation}
  unchanged. Therefore $\recoveryfact$ is non-identified.
\end{theorem}

\begin{remark}[\hdr{recov:scope}]
  The non-identification is necessary inside the additive frame-offset model; the model
  itself is feasible.
\end{remark}

\begin{insight}[\hdr{recov:victim}]\grade{feas}
  Silence routes to the contested anchor: absent external access, the victim's report is
  another internally framed observation, not a portable certificate.
\end{insight}

\begin{proposition}[\hdr{recov:ancill}]\grade{feas}
  Since
  \begin{equation}
    \effectivereads[\ell]
    =\frac{\cleanreads[\ell]\infopool[\ell]}
      {\cleanreads[\ell]+\infopool[\ell]},
    \label{eq:recov:effharm}
  \end{equation}
  its margins are
  \begin{equation}
    \frac{\partial\effectivereads[\ell]}{\partial\cleanreads[\ell]}
    =\frac{\infopool[\ell]^2}{(\cleanreads[\ell]+\infopool[\ell])^2}
    \to0,
    \label{eq:recov:margclean}
  \end{equation}
  while
  \begin{equation}
    \frac{\partial\effectivereads[\ell]}{\partial\infopool[\ell]}
    =\frac{\cleanreads[\ell]^2}{(\cleanreads[\ell]+\infopool[\ell])^2}
    \to1.
    \label{eq:recov:marginfo}
  \end{equation}
  Another reader dies at the margin; an independent anchor moves the cap. The limits
  name their regime: the reader margin dies where readers are abundant, and the anchor
  margin approaches one only in the anchor-\emph{scarce} corner --- with anchors
  abundant it dies the same way. The defensible sentence is symmetric: the margin of
  whichever factor is scarce stays high.
\end{proposition}

\begin{proposition}[\hdr{recov:ceil}]\grade{feas}
  Under covering, any one-place certificate is bounded by the anchor ceiling
  \begin{equation}
    \precision\sb{\mathrm{certificate}}
    \le
    \mathbf 1\transpose
    \pinv{\frameloading\anchorprior\frameloading\transpose}\mathbf 1.
    \label{eq:recov:certceil}
  \end{equation}
  This is a supporting prohibition, not a flagship falsifier; the falsifiable standing
  lives in the skew sector (\ftag{dyn:pitchpair}, \ftag{dyn:rot}).
\end{proposition}

\begin{remark}[\hdr{recov:auditor}]
  Testing the ceiling requires independently varying readers and anchors. Therefore the test
  requires an external auditor.
\end{remark}

\begin{theorem}[\hdr{recov:compare}]\grade{feas}
  For multiple subjects,
  \begin{equation}
    x_e=\mu_{k(e)}+z_e\transpose\frameoffsets+\varepsilon_e.
    \label{eq:recov:compare}
  \end{equation}
  Build the bipartite graph whose nodes are subjects $k$ and frames $\ell$, with an edge when
  subject $k$ is observed through frame $\ell$. Then
  \begin{equation}
    \boxed{
      \begin{gathered}
      \mu_k-\mu_{k'}\text{ is identified}
      \quad\Longleftrightarrow\quad\\
      k,k'\text{ lie in the same connected component}.
      \end{gathered}
    }
    \label{eq:recov:component}
  \end{equation}
  Absolute levels remain unidentified up to one constant per component.
  The result is the two-way fixed-effects connectedness condition
  \parencite{a:akm}, transported: subjects for workers, frames for firms, and the
  same moral --- contrasts identify within a connected component, levels never.
\end{theorem}

\begin{remark}[\hdr{recov:groupframe}]
  If offsets are group-keyed as $\groupoffset{\ell}{g(k)}$, cross-group contrasts require
  bridge anchors.
\end{remark}

\begin{remark}[\hdr{recov:contrast}]
  A contrast is two-place. A certificate is one-place. Comparative recovery does not revive
  the portable verdict.
\end{remark}

\begin{proposition}[\hdr{recov:anchorflow}]\grade{feas}
  Let
  \begin{equation}
    \anchorprecision:=b\sb{\ell}\sp{-2}
    \qquad \dimpublicwelfareinvsq.
    \label{eq:recov:anchorprec}
  \end{equation}
  Write
  \begin{equation}
    \dot p_\ell=\anchorflux-\anchorlapse p_\ell.
    \label{eq:recov:anchorode}
  \end{equation}
  At rest, $\anchorflux=\anchorlapse p_\ell^\star$. Without maintenance,
  \begin{equation}
    p_\ell(t)=p_\ell(0)e^{-\anchorlapse t}.
    \label{eq:recov:anchordecay}
  \end{equation}
\end{proposition}

\begin{insight}[\hdr{recov:hold}]\grade{feas}
  Anchor precision is held by maintenance, not merely observed.
\end{insight}

\begin{insight}[\hdr{recov:held}]\grade{feas}
  A verdict is held, not had.
\end{insight}

\begin{theorem}[\hdr{recov:gap}]\grade{nec}
  The recovery appendix asks for three independence objects --- the witness Gram;
  $\idcov$ versus $\frameloading\anchorprior\frameloading\transpose$; and
  $\audiencediversity$ versus true anchors --- and all reduce to the same external wall.
\end{theorem}

\end{statement}

\subsection{The evidential register}
\label{subsec:ledger:evidence}
Derivation is a DAG; evidence is not. The two basins meet once, at the flow
(\ftag{gen:basins}), and their contact is WARRANT rather than entailment: every
back-edge is posted here, and an edge this register does not carry is an edge the
document does not lean on.
