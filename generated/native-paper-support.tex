\msg_new:nnn { ftoo } { hdr-two-argument-form }
  {
    \token_to_str:N \hdr~takes~the~tag~alone:~the~heading~is~read~from~the~paper's~
    manifest~data.~The~second~argument~seen~here~was~'#1'.
  }
\msg_new:nnn { ftoo } { hdr-unbound-tag }
  {
    The~result~head~mints~'#1',~a~tag~the~manifest~data~binds~to~no~referent.~
    Bind~'#1'~in~documents/paper/data/ftoo_manifest.tex~or~do~not~mint~it.
  }
\msg_new:nnn { ftoo } { hdr-alias-tag }
  {
    The~result~head~mints~'#1',~which~is~an~alias~rather~than~the~canonical~
    tag~of~its~referent.~A~mint~must~print~and~anchor~one~published~name.
  }
\msg_new:nnn { ftoo } { hdr-no-heading }
  {
    '#1'~resolves~to~a~referent~the~manifest~data~publishes~no~heading~for.~The~
    heading~is~missing;~review~the~tag~and~its~manifest~declaration.
  }

  }
